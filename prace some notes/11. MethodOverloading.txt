public class MethodOverloading {
    
    static int add(int a, int b) {
        return a + b;
    }
    
    static int add(int a, int b, int c) {
        return a + b + c;
    }
    
    public static void main(String[] args) {
        int sum1 = add(5, 10);
        int sum2 = add(2, 4, 6);
        
        System.out.println("Sum 1: " + sum1);
        System.out.println("Sum 2: " + sum2);
    }
}
